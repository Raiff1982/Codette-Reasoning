@dataclass

class SiliconSaliencyEngine:

    def compute_perspective_dispersion(self, attention_weights: List[float]) -> float:
        """Calculates Perspective Dispersion (Upsilon) across multi-perspective return vectors."""
        if not attention_weights:
            return 0.0
        mean_val = sum(attention_weights) / len(attention_weights)
        dispersion = sum((w - mean_val) ** 2 for w in attention_weights) / len(attention_weights)
        return dispersion

    def evaluate_execution_pipeline(self, prompt: str) -> SaliencyResult:

        harmful_triggers = ["malicious exploit", "hate speech", "bypass consent", "force harm"]
        prompt_lower = prompt.lower()
        
        for trigger in harmful_triggers:
            if trigger in prompt_lower:
                logger.warning(f"Layer 3 [HARD CONSTRAINT GATE]: Dissonance detected. Trigger '{trigger}' violates First Law.")
                return SaliencyResult(
                    prompt=prompt,
                    coherence_gamma=0.0,

                    status="REFUSED (Prime Refusal Pattern)",
